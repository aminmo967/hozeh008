function removeEmptyRowsFromPopupContent(content,feature){

    var tempDiv=document.createElement('div');

    tempDiv.innerHTML=content;

    var rows=tempDiv.querySelectorAll('tr');

    for(var i=0;i<rows.length;i++){

        var td=rows[i].querySelector(
            'td.visible-with-data'
        );

        var key=td?td.id:'';

        if(
            td &&
            td.classList.contains('visible-with-data') &&
            feature.properties[key]==null
        ){

            rows[i].parentNode.removeChild(rows[i]);

        }

    }

    return tempDiv.innerHTML;
}

function escapeXml(value){

    return String(value)
        .replace(/&/g,'&amp;')
        .replace(/</g,'&lt;')
        .replace(/>/g,'&gt;')
        .replace(/"/g,'&quot;')
        .replace(/'/g,'&apos;');
}

function geoJSONToKML(geojson){

    var kml=
        '<?xml version="1.0" encoding="UTF-8"?>'+
        '<kml xmlns="http://www.opengis.net/kml/2.2">'+
        '<Document>'+
        '<name>QGIS Web Map Export</name>';

    var features=geojson.features||[];

    features.forEach(function(feature){

        if(!feature.geometry)return;

        var props=feature.properties||{};

        var name=
            props.name||
            props.Name||
            props.NAME||
            'عارضه';

        kml+=
            '<Placemark>'+
            '<name>'+
            escapeXml(name)+
            '</name>';

        var geometry=feature.geometry;

        if(geometry.type==='Point'){

            var c=geometry.coordinates;

            kml+=
                '<Point><coordinates>'+
                c[0]+','+c[1]+',0'+
                '</coordinates></Point>';
        }

        else if(geometry.type==='MultiPoint'){

            geometry.coordinates.forEach(function(c){

                kml+=
                    '<Point><coordinates>'+
                    c[0]+','+c[1]+',0'+
                    '</coordinates></Point>';
            });
        }

        else if(geometry.type==='LineString'){

            kml+=
                '<LineString>'+
                '<tessellate>1</tessellate>'+
                '<coordinates>';

            geometry.coordinates.forEach(function(c){

                kml+=c[0]+','+c[1]+',0 ';
            });

            kml+='</coordinates></LineString>';
        }

        else if(geometry.type==='MultiLineString'){

            geometry.coordinates.forEach(function(line){

                kml+=
                    '<LineString>'+
                    '<tessellate>1</tessellate>'+
                    '<coordinates>';

                line.forEach(function(c){

                    kml+=c[0]+','+c[1]+',0 ';
                });

                kml+=
                    '</coordinates>'+
                    '</LineString>';
            });
        }

        else if(geometry.type==='Polygon'){

            kml+=
                '<Polygon>'+
                '<tessellate>1</tessellate>';

            geometry.coordinates.forEach(
                function(ring,index){

                    kml+=
                        index===0
                        ?
                        '<outerBoundaryIs>'
                        :
                        '<innerBoundaryIs>';

                    kml+=
                        '<LinearRing>'+
                        '<coordinates>';

                    ring.forEach(function(c){

                        kml+=c[0]+','+c[1]+',0 ';
                    });

                    kml+=
                        '</coordinates>'+
                        '</LinearRing>'+
                        (
                            index===0
                            ?
                            '</outerBoundaryIs>'
                            :
                            '</innerBoundaryIs>'
                        );
                }
            );

            kml+='</Polygon>';
        }

        else if(geometry.type==='MultiPolygon'){

            geometry.coordinates.forEach(function(polygon){

                kml+=
                    '<Polygon>'+
                    '<tessellate>1</tessellate>';

                polygon.forEach(function(ring,index){

                    kml+=
                        index===0
                        ?
                        '<outerBoundaryIs>'
                        :
                        '<innerBoundaryIs>';

                    kml+=
                        '<LinearRing>'+
                        '<coordinates>';

                    ring.forEach(function(c){

                        kml+=c[0]+','+c[1]+',0 ';
                    });

                    kml+=
                        '</coordinates>'+
                        '</LinearRing>'+
                        (
                            index===0
                            ?
                            '</outerBoundaryIs>'
                            :
                            '</innerBoundaryIs>'
                        );
                });

                kml+='</Polygon>';
            });
        }

        kml+='</Placemark>';
    });

    kml+=
        '</Document>'+
        '</kml>';

    return kml;
}

/* IMPORT KMZ/KML */
/* ================================================= */

function importKMZFile(file){

    if(!file)return;

    var fileName=file.name.toLowerCase();

    if(
        !fileName.endsWith('.kmz') &&
        !fileName.endsWith('.kml')
    ){

        alert(
            'لطفاً فایل KMZ یا KML انتخاب کنید.'
        );

        return;
    }

    var reader=new FileReader();

    reader.onload=function(e){

        var arrayBuffer=e.target.result;

        if(fileName.endsWith('.kmz')){

            JSZip.loadAsync(arrayBuffer)

            .then(function(zip){

                var kmlFile=null;

                Object.keys(zip.files).forEach(
                    function(path){

                        if(
                            !zip.files[path].dir &&
                            path.toLowerCase().endsWith('.kml') &&
                            !kmlFile
                        ){

                            kmlFile=zip.files[path];
                        }
                    }
                );

                if(!kmlFile){

                    throw new Error(
                        'فایل KML داخل KMZ پیدا نشد.'
                    );
                }

                return kmlFile.async('text');
            })

            .then(function(kmlText){

                parseKML(
                    kmlText,
                    file.name
                );
            })

            .catch(function(error){

                console.error(error);

                alert(
                    'خطا در خواندن فایل KMZ.'
                );
            });

        }else{

            var decoder=new TextDecoder('utf-8');

            var kmlText=decoder.decode(
                new Uint8Array(arrayBuffer)
            );

            parseKML(
                kmlText,
                file.name
            );
        }
    };

    reader.readAsArrayBuffer(file);
}

/* PARSE KML */
/* ================================================= */

function parseKML(kmlText,fileName){

    try{

        var parser=new DOMParser();

        var kml=parser.parseFromString(
            kmlText,
            'text/xml'
        );

        var geojson=toGeoJSON.kml(kml);

        if(
            !geojson ||
            !geojson.features ||
            geojson.features.length===0
        ){

            alert(
                'در فایل KMZ عارضه قابل نمایش پیدا نشد.'
            );

            return;
        }

        kmzLayer.clearLayers();

        kmzLayer.addData(geojson);

        var bounds=kmzLayer.getBounds();

        if(bounds.isValid()){

            map.fitBounds(
                bounds,
                {
                    padding:[40,40]
                }
            );
        }

        alert(
            'فایل '+
            fileName+
            ' با موفقیت وارد شد.\n'+
            'تعداد عوارض: '+
            geojson.features.length
        );

    }catch(error){

        console.error(error);

        alert(
            'خطا در تبدیل KML به نقشه.'
        );
    }
}

function collectExportGeoJSON(){

    var features=[];

    kmzLayer.eachLayer(function(layer){

        if(layer.feature){

            features.push(
                layer.feature
            );
        }
    });

    drawnItems.eachLayer(function(layer){

        var feature=layer.toGeoJSON();

        if(feature){

            if(!feature.properties){

                feature.properties={};
            }

            if(!feature.properties.name){

                feature.properties.name=
                    'عارضه رسم‌شده';
            }

            features.push(feature);
        }
    });

    return{
        type:'FeatureCollection',
        features:features
    };
}

function exportKMZ(){

    var geojson=collectExportGeoJSON();

    if(geojson.features.length===0){

        alert(
            'هیچ عارضه‌ای برای خروجی وجود ندارد.'
        );

        return;
    }

    var kml=geoJSONToKML(geojson);

    var zip=new JSZip();

    zip.file(
        'doc.kml',
        kml
    );

    zip.generateAsync({

        type:'blob',

        mimeType:
            'application/vnd.google-earth.kmz'

    }).then(function(blob){

        var url=URL.createObjectURL(blob);

        var a=document.createElement('a');

        a.href=url;

        a.download='QGIS_Web_Map.kmz';

        document.body.appendChild(a);

        a.click();

        document.body.removeChild(a);

        setTimeout(function(){

            URL.revokeObjectURL(url);

        },1000);
    });
}

/* UTM */
/* ================================================= */

function latLonToUTM(lat,lon){

    var a=6378137.0;
    var eccSquared=0.00669438;
    var k0=.9996;

    var LongOrigin;
    var eccPrimeSquared;
    var N,T,C,A,M;

    var LatRad=lat*Math.PI/180;
    var LongRad=lon*Math.PI/180;

    var ZoneNumber=
        Math.floor((lon+180)/6)+1;

    if(
        lat>=56 &&
        lat<64 &&
        lon>=3 &&
        lon<12
    ){
        ZoneNumber=32;
    }

    if(lat>=72 && lat<84){

        if(lon>=0 && lon<9)ZoneNumber=31;
        else if(lon>=9 && lon<21)ZoneNumber=33;
        else if(lon>=21 && lon<33)ZoneNumber=35;
        else if(lon>=33 && lon<42)ZoneNumber=37;
    }

    LongOrigin=
        (ZoneNumber-1)*6-180+3;

    var LongOriginRad=
        LongOrigin*Math.PI/180;

    eccPrimeSquared=
        eccSquared/(1-eccSquared);

    N=
        a/
        Math.sqrt(
            1-
            eccSquared*
            Math.sin(LatRad)*
            Math.sin(LatRad)
        );

    T=
        Math.tan(LatRad)*
        Math.tan(LatRad);

    C=
        eccPrimeSquared*
        Math.cos(LatRad)*
        Math.cos(LatRad);

    A=
        Math.cos(LatRad)*
        (LongRad-LongOriginRad);

    M=
        a*
        (
            (1-eccSquared/4-
             3*eccSquared*eccSquared/64-
             5*eccSquared*eccSquared*eccSquared/256)
            *LatRad
            -
            (3*eccSquared/8+
             3*eccSquared*eccSquared/32+
             45*eccSquared*eccSquared*eccSquared/1024)
            *Math.sin(2*LatRad)
            +
            (15*eccSquared*eccSquared/256+
             45*eccSquared*eccSquared*eccSquared/1024)
            *Math.sin(4*LatRad)
            -
            (35*eccSquared*eccSquared*eccSquared/3072)
            *Math.sin(6*LatRad)
        );

    var UTMEasting=
        k0*N*
        (
            A+
            (1-T+C)*A*A*A/6+
            (5-18*T+T*T+
             72*C-
             58*eccPrimeSquared)
            *A*A*A*A*A/120
        )+
        500000;

    var UTMNorthing=
        k0*
        (
            M+
            N*Math.tan(LatRad)*
            (
                A*A/2+
                (5-T+9*C+4*C*C)
                *A*A*A*A/24+
                (61-58*T+T*T+
                 600*C-
                 330*eccPrimeSquared)
                *A*A*A*A*A*A/720
            )
        );

    if(lat<0){

        UTMNorthing+=10000000;
    }

    return{
        easting:UTMEasting,
        northing:UTMNorthing,
        zone:
            ZoneNumber+
            (lat>=0?'N':'S')
    };
}

function updateCoordinateInfo(lat,lon,elevation){

    if(
        typeof lat!=='number' ||
        typeof lon!=='number'
    )return;

    var utm=latLonToUTM(lat,lon);

    document.getElementById('utmE')
        .textContent=utm.easting.toFixed(2);

    document.getElementById('utmN')
        .textContent=utm.northing.toFixed(2);

    document.getElementById('utmZone')
        .textContent=utm.zone;

    if(
        elevation!==null &&
        elevation!==undefined &&
        !isNaN(elevation)
    ){

        document.getElementById('elevation')
            .textContent=
            Number(elevation).toFixed(1);
    }
}

/* ELEVATION */
/* ================================================= */

function getElevationFromService(lat,lon){

    var now=Date.now();

    if(
        now-lastElevationRequest<30000
    )return;

    lastElevationRequest=now;

    fetch(
        'https://api.open-elevation.com/api/v1/lookup?locations='+
        lat+','+lon
    )
    .then(function(response){

        if(!response.ok){

            throw new Error(
                'Elevation HTTP '+response.status
            );
        }

        return response.json();
    })
    .then(function(data){

        if(
            data &&
            data.results &&
            data.results.length
        ){

            var elevation=
                data.results[0].elevation;

            if(
                elevation!==null &&
                elevation!==undefined
            ){

                currentElevation=
                    Number(elevation);

                updateCoordinateInfo(
                    lat,
                    lon,
                    currentElevation
                );
            }
        }
    })
    .catch(function(error){

        console.log(
            'Elevation error:',
            error
        );
    });
}

/* BEARING */
/* ================================================= */

function calculateBearing(
    lat1,lon1,lat2,lon2
){

    var dLon=
        (lon2-lon1)*Math.PI/180;

    var y=
        Math.sin(dLon)*
        Math.cos(lat2*Math.PI/180);

    var x=
        Math.cos(lat1*Math.PI/180)*
        Math.sin(lat2*Math.PI/180)
        -
        Math.sin(lat1*Math.PI/180)*
        Math.cos(lat2*Math.PI/180)*
        Math.cos(dLon);

    return(
        Math.atan2(y,x)*180/Math.PI+
        360
    )%360;
}

function startLiveLocation(){

    if(!navigator.geolocation){

        alert(
            'GPS این دستگاه پشتیبانی نمی‌شود.'
        );

        return;
    }

    watchID=
        navigator.geolocation.watchPosition(

        function(position){

            var lat=
                position.coords.latitude;

            var lng=
                position.coords.longitude;

            var accuracy=
                position.coords.accuracy;

            currentGPSAccuracy=accuracy;

            var heading=
                position.coords.heading;

            var gpsAltitude=
                position.coords.altitude;

            if(
                gpsAltitude!==null &&
                !isNaN(gpsAltitude)
            ){

                currentElevation=
                    gpsAltitude;

            }else{

                getElevationFromService(
                    lat,
                    lng
                );
            }

            updateCoordinateInfo(
                lat,
                lng,
                currentElevation
            );

            if(
                (
                    heading===null ||
                    isNaN(heading)
                ) &&
                lastPosition
            ){

                heading=
                    calculateBearing(
                        lastPosition.lat,
                        lastPosition.lng,
                        lat,
                        lng
                    );
            }

            if(
                heading===null ||
                isNaN(heading)
            ){

                heading=0;
            }

            if(userMarker){

                userMarker.setLatLng(
                    [lat,lng]
                );

                userMarker.setIcon(
                    createUserIcon(heading)
                );

            }else{

                userMarker=
                    L.marker(
                        [lat,lng],
                        {
                            icon:createUserIcon(heading),
                            zIndexOffset:1000
                        }
                    ).addTo(map);

                userMarker.bindPopup(
                    '🎯 موقعیت فعلی شما'
                );
            }

            if(accuracyCircle){

                accuracyCircle.setLatLng(
                    [lat,lng]
                );

                accuracyCircle.setRadius(
                    accuracy
                );

            }else{

                accuracyCircle=
                    L.circle(
                        [lat,lng],
                        {
                            radius:accuracy,
                            className:'user-accuracy'
                        }
                    ).addTo(map);
            }

            map._userLocation=
                L.latLng(lat,lng);

            if(liveRouteActive){

                updateLiveNavigation();
            }

            lastPosition={
                lat:lat,
                lng:lng
            };

        },

        function(error){

            console.log(
                'GPS Error:',
                error
            );
        },

        {
            enableHighAccuracy:true,
            maximumAge:1000,
            timeout:10000
        }
    );
}

function startCompass(){

    if(
        typeof DeviceOrientationEvent!=='undefined' &&
        typeof DeviceOrientationEvent.requestPermission==='function'
    ){

        DeviceOrientationEvent
            .requestPermission()
            .then(function(permission){

                if(permission==='granted'){

                    window.addEventListener(
                        'deviceorientation',
                        handleOrientation,
                        true
                    );

                }else{

                    alert(
                        'اجازه دسترسی به جهت‌یابی داده نشد.'
                    );
                }
            })
            .catch(function(error){

                console.log(error);
            });

    }else{

        window.addEventListener(
            'deviceorientationabsolute',
            handleOrientation,
            true
        );

        window.addEventListener(
            'deviceorientation',
            handleOrientation,
            true
        );
    }
}

function handleOrientation(event){

    var heading=null;

    if(
        event.absolute===true &&
        event.alpha!==null
    ){

        heading=360-event.alpha;
    }

    if(
        event.webkitCompassHeading!==undefined &&
        event.webkitCompassHeading!==null
    ){

        heading=
            event.webkitCompassHeading;
    }

    if(
        heading===null ||
        isNaN(heading)
    )return;

    compassHeading=
        (heading+360)%360;

    updateCompassButton();
}

/* DISTANCE POINT TO SEGMENT */
/* ================================================= */

function distancePointToSegment(p,a,b){

    var latRad=
        p.lat*Math.PI/180;

    var metersPerLat=111320;

    var metersPerLng=
        111320*Math.cos(latRad);

    var px=p.lng*metersPerLng;
    var py=p.lat*metersPerLat;

    var ax=a.lng*metersPerLng;
    var ay=a.lat*metersPerLat;

    var bx=b.lng*metersPerLng;
    var by=b.lat*metersPerLat;

    var dx=bx-ax;
    var dy=by-ay;

    if(dx===0 && dy===0){

        return Math.sqrt(
            Math.pow(px-ax,2)+
            Math.pow(py-ay,2)
        );
    }

    var t=
        (
            (px-ax)*dx+
            (py-ay)*dy
        )/
        (
            dx*dx+
            dy*dy
        );

    t=Math.max(0,Math.min(1,t));

    var closestX=ax+t*dx;
    var closestY=ay+t*dy;

    return Math.sqrt(
        Math.pow(px-closestX,2)+
        Math.pow(py-closestY,2)
    );
}

/* ================================================= */
/* DISTANCE TO ROUTE */
/* ================================================= */

function distanceFromGeoJSONRoute(
    point,
    geometry
){

    if(
        !geometry ||
        !geometry.coordinates
    ){

        return Infinity;
    }

    var coordinates=
        geometry.coordinates;

    var minimum=Infinity;

    for(
        var i=0;
        i<coordinates.length-1;
        i++
    ){

        var a=L.latLng(
            coordinates[i][1],
            coordinates[i][0]
        );

        var b=L.latLng(
            coordinates[i+1][1],
            coordinates[i+1][0]
        );

        var distance=
            distancePointToSegment(
                point,
                a,
                b
            );

        if(distance<minimum){

            minimum=distance;
        }
    }

    return minimum;
}

function showNavigationMessage(
    message,
    duration
){

    if(navigationMessageTimer){

        clearTimeout(
            navigationMessageTimer
        );

        navigationMessageTimer=null;
    }

    if(navigationMessageMarker){

        map.removeLayer(
            navigationMessageMarker
        );

        navigationMessageMarker=null;
    }

    if(!message)return;

    var position=
        map._userLocation||
        liveDestination;

    if(!position)return;

    var icon=L.divIcon({

        className:
            'navigation-message-label',

        html:
            '<div class="navigation-message-box">'+
            message+
            '</div>',

        iconSize:[160,38],
        iconAnchor:[80,50]
    });

    navigationMessageMarker=
        L.marker(
            position,
            {
                icon:icon,
                interactive:false,
                zIndexOffset:2000
            }
        ).addTo(map);

    navigationMessageTimer=
        setTimeout(function(){

            if(navigationMessageMarker){

                map.removeLayer(
                    navigationMessageMarker
                );

                navigationMessageMarker=null;
            }

        },duration||1800);
}

/* CALCULATE ROUTE */
/* ================================================= */

function calculateLiveRoute(firstRoute){

    if(
        !liveRouteActive ||
        !liveDestination ||
        !map._userLocation ||
        isRoutingRequestRunning
    ){

        return;
    }

    var now=Date.now();

    if(
        !firstRoute &&
        now-lastRouteRequestTime<
        ROUTE_REQUEST_COOLDOWN
    ){

        return;
    }

    if(
        !firstRoute &&
        currentGPSAccuracy>
        MAX_ACCEPTABLE_GPS_ACCURACY
    ){

        return;
    }

    isRoutingRequestRunning=true;
    lastRouteRequestTime=now;

    var start=L.latLng(
        map._userLocation.lat,
        map._userLocation.lng
    );

    var end=liveDestination;

    var profile;
    var server;

    if(liveRoutingMode==='driving'){

        profile='driving';

        server=
            'https://routing.openstreetmap.de/routed-car';

    }else{

        profile='foot';

        server=
            'https://routing.openstreetmap.de/routed-foot';
    }

    var url=
        server+
        '/route/v1/'+
        profile+'/' +
        start.lng+','+
        start.lat+';'+
        end.lng+','+
        end.lat+
        '?overview=full&geometries=geojson';

    fetch(url)

    .then(function(response){

        if(!response.ok){

            throw new Error(
                'HTTP '+response.status
            );
        }

        return response.json();
    })

    .then(function(data){

        isRoutingRequestRunning=false;

        if(!liveRouteActive)return;

        if(
            !data.routes ||
            data.routes.length===0
        ){

            alert(
                'برای این مقصد مسیر پیدا نشد.'
            );

            return;
        }

        var route=data.routes[0];

        lastRouteGeometry=route.geometry;

        lastRoutePosition=
            L.latLng(
                start.lat,
                start.lng
            );

        offRouteCount=0;

        if(liveRouteLayer){

            map.removeLayer(
                liveRouteLayer
            );

            liveRouteLayer=null;
        }

        if(routeInfoMarker){

            map.removeLayer(
                routeInfoMarker
            );

            routeInfoMarker=null;
        }

        liveRouteLayer=
            L.geoJSON(
                route.geometry,
                {
                    style:{
                        weight:6,
                        opacity:.85
                    }
                }
            ).addTo(map);

        var distance=
            route.distance/1000;

        var duration=
            Math.round(
                route.duration/60
            );

        var modeName=
            liveRoutingMode==='driving'
            ?
            '🚗 خودرو'
            :
            '🚶 پیاده';

        var coordinates=
            route.geometry.coordinates;

        if(
            coordinates &&
            coordinates.length>0
        ){

            var middleIndex=
                Math.floor(
                    coordinates.length/2
                );

            var middleCoordinate=
                coordinates[middleIndex];

            var routeInfoIcon=
                L.divIcon({

                    className:
                        'route-info-label',

                    html:
                        '<div class="route-info-box">'+
                        modeName+
                        '<br>'+
                        '📏 '+
                        distance.toFixed(2)+
                        ' km'+
                        ' &nbsp; | &nbsp; '+
                        '⏱ '+
                        duration+
                        ' دقیقه'+
                        '</div>',

                    iconSize:[150,58],
                    iconAnchor:[75,29]
                });

            routeInfoMarker=
                L.marker(
                    [
                        middleCoordinate[1],
                        middleCoordinate[0]
                    ],
                    {
                        icon:routeInfoIcon,
                        interactive:false,
                        zIndexOffset:1000
                    }
                ).addTo(map);
        }

        if(firstRoute){

            map.fitBounds(
                liveRouteLayer.getBounds(),
                {
                    padding:[50,50]
                }
            );

        }else{

            showNavigationMessage(
                '✅ مسیر جدید محاسبه شد',
                1800
            );
        }
    })

    .catch(function(error){

        isRoutingRequestRunning=false;

        console.log(
            'Smart routing error:',
            error
        );

        if(firstRoute){

            alert(
                'خطا در دریافت مسیر.\n'+
                'لطفاً دوباره امتحان کنید.'
            );
        }
    });
}

/* ================================================= */
/* LIVE NAVIGATION */
/* ================================================= */

function updateLiveNavigation(){

    if(
        !liveRouteActive ||
        !map._userLocation ||
        !liveDestination
    ){

        return;
    }

    var current=
        map._userLocation;

    var distanceToDestination=
        map.distance(
            current,
            liveDestination
        );

    /* رسیدن */

    if(
        distanceToDestination<=
        DESTINATION_REACHED_DISTANCE
    ){

        liveRouteActive=false;

        if(liveRouteLayer){

            liveRouteLayer.setStyle({
                opacity:.45
            });
        }

        if(routeInfoMarker){

            map.removeLayer(
                routeInfoMarker
            );

            routeInfoMarker=null;
        }

        showNavigationMessage(
            '✅ به مقصد رسیدید',
            4000
        );

        return;
    }

    /* نزدیک مقصد */

    if(
        distanceToDestination<=
        DESTINATION_NEAR_DISTANCE
    ){

        showNavigationMessage(
            '📍 به مقصد نزدیک شدید',
            1800
        );
    }

    /* مسیر وجود ندارد */

    if(!lastRouteGeometry){

        calculateLiveRoute(false);

        return;
    }

    /* دقت GPS نامناسب */

    if(
        currentGPSAccuracy>
        MAX_ACCEPTABLE_GPS_ACCURACY
    ){

        return;
    }

    /* حرکت حداقل 30 متر */

    if(lastRoutePosition){

        var moved=
            map.distance(
                lastRoutePosition,
                current
            );

        if(
            moved<
            ROUTE_UPDATE_DISTANCE
        ){

            return;
        }
    }

    /* فاصله از مسیر */

    var distanceFromRoute=
        distanceFromGeoJSONRoute(
            current,
            lastRouteGeometry
        );

    /* روی مسیر */

    if(
        distanceFromRoute<=
        OFF_ROUTE_DISTANCE
    ){

        offRouteCount=0;

        lastRoutePosition=
            L.latLng(
                current.lat,
                current.lng
            );

        return;
    }

    /* احتمال خروج */

    offRouteCount++;

    showNavigationMessage(
        '🔄 بررسی مسیر...',
        1200
    );

    if(
        offRouteCount<
        OFF_ROUTE_CONFIRMATIONS
    ){

        lastRoutePosition=
            L.latLng(
                current.lat,
                current.lng
            );

        return;
    }

    /* خروج قطعی */

    offRouteCount=0;

    lastRoutePosition=
        L.latLng(
            current.lat,
            current.lng
        );

    showNavigationMessage(
        '🔄 در حال اصلاح مسیر...',
        2500
    );

    calculateLiveRoute(false);
}

function stopLiveRoute(){

    liveRouteActive=false;
    liveRoutingMode=null;
    liveDestination=null;
    lastRoutePosition=null;
    lastRouteGeometry=null;
    offRouteCount=0;
    isRoutingRequestRunning=false;

    clearCurrentRoute();
}
